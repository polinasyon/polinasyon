<script>
const uygulama = {
    data: { konum: "Ankara", lat: 39.92, lon: 32.85 },

    async otomatikKonumAl() {
        if (!navigator.geolocation) return alert("Tarayıcın konum desteği yok.");
        
        try {
            const pos = await new Promise((res, rej) => navigator.geolocation.getCurrentPosition(res, rej));
            const lat = pos.coords.latitude;
            const lon = pos.coords.longitude;

            const res2 = await fetch(`https://api.bigdatacloud.net/data/reverse-geocode-client?latitude=${lat}&longitude=${lon}&localityLanguage=tr`);
            const yer = await res2.json();

            const sehir = yer.city || yer.locality || "Yeni Konum";
            document.getElementById('havaLokasyon').textContent = sehir;
            alert(`✅ Konum alındı: ${sehir}`);
        } catch(e) {
            alert("Konum alınamadı. Manuel deneyin.");
        }
    },

    konumModaliniAc() {
        const sehir = prompt("İl veya İlçe girin (örnek: Muğla, Rize, Simav):", this.data.konum);
        if (sehir) {
            document.getElementById('havaLokasyon').textContent = sehir;
            this.data.konum = sehir;
        }
    },

    verileriYedekle() {
        alert("Yedek alma özelliği yakında eklenecek.");
    }
};

window.onload = () => {
    lucide.createIcons();
    uygulama.otomatikKonumAl();   // İlk açılışta otomatik dene
};
</script>
