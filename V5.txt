            nectar_flow_score(temp, humidity, wind_speed) {
                const temp_score = Math.max(0, 1 - Math.abs(temp - 22) / 15);
                const hum_score = Math.max(0, 1 - Math.abs(humidity - 70) / 30);
                const wind_score = Math.max(0, 1 - wind_speed / 20);
                const total = (temp_score * 0.4 + hum_score * 0.3 + wind_score * 0.3) * 100;
                return Math.round(total);
            },

                this.data.sahaGozlemleri.push({ tarih, tur, bitki, not });
                this.veriKaydet();
                
                this.sahaRaporuModaliniKapat();
                this.timelineCiz();
            },

            async havaDurumunuYukle(lat, lon, sehirAd) {
                document.getElementById('havaLokasyon').textContent = sehirAd;
                this.data.konum = sehirAd;
                this.data.lat = lat;
                this.data.lon = lon;

                try {
                    const res = await fetch(
                        `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current=temperature_2m,relative_humidity_2m,wind_speed_10m&timezone=auto&models=icon_eu,ecmwf_ifs04,best_match`,
                        { cache: 'no-cache' }
                    );
                    const data = await res.json();
                    
                    this.data.temp = Math.round(data.current.temperature_2m);
                    this.data.humidity = Math.round(data.current.relative_humidity_2m); 
                    this.data.windSpeed = Math.round(data.current.wind_speed_10m);

                    document.getElementById('havaSicaklik').textContent = `${this.data.temp}°C`;
                    document.getElementById('havaNem').textContent = `%${this.data.humidity}`;
                    document.getElementById('havaRuzgar').textContent = `${this.data.windSpeed} km/s`;

                    // Çevrimdışı kullanım için son hava verilerini yerel hafızaya sakla
                    localStorage.setItem('polinasyon_last_weather', JSON.stringify({
                        temp: this.data.temp,
                        humidity: this.data.humidity,
                        windSpeed: this.data.windSpeed,
                        time: new Date().toLocaleTimeString('tr-TR', { hour: '2-digit', minute: '2-digit' })
                    }));

                } catch(e) {
                    console.warn("Canlı hava verisi çekilemedi, önbelleğe bakılıyor: ", e);
                    
                    // İnternet olmaması durumunda en son kaydedilen veriyi getir
                    const sonHava = localStorage.getItem('polinasyon_last_weather');
                    if (sonHava) {
                        try {
                            const parsed = JSON.parse(sonHava);
                            this.data.temp = parsed.temp;
                            this.data.humidity = parsed.humidity;
                            this.data.windSpeed = parsed.windSpeed;
                            
                            document.getElementById('havaSicaklik').textContent = `${this.data.temp}°C`;
                            document.getElementById('havaNem').textContent = `%${this.data.humidity}`;
                            document.getElementById('havaRuzgar').textContent = `${this.data.windSpeed} km/s`;
                        } catch(err) {
                            console.error("Önbellekteki hava durumu okunamadı", err);
                        }
                    }
                }

                this.veriKaydet(); 
                this.nektarAkimiHesaplaVeGuncelle();
                this.timelineCiz();
            },

            arayuzuGuncelle() {
                const kategoriContainer = document.getElementById('kategoriListesiContainer');
                if (!this.data.kategoriler || this.data.kategoriler.length === 0) {
                    kategoriContainer.innerHTML = `<p class="text-center py-8 text-slate-400 text-xs">Henüz kategori bulunmuyor.</p>`;
                    return;
                }

                let html = '';
                this.data.kategoriler.forEach(kat => {
                    const kovanSayisi = kat.kovanlar ? kat.kovanlar.length : 0;
                    html += `
                        <div onclick="uygulama.kategoriSec(${kat.id})" class="soft-card p-4 cursor-pointer hover:border-amber-400 flex items-center justify-between group transition">
                            <div class="flex items-center gap-3.5">
                                <div class="w-11 h-11 rounded-2xl bg-amber-50 group-hover:bg-amber-500 group-hover:text-white text-amber-600 flex items-center justify-center transition shrink-0">
                                    <i data-lucide="folder-kanban" class="w-5 h-5"></i>
                                </div>
                                <div>
                                    <h3 class="font-bold text-slate-800 text-base">${kat.isim}</h3>
                                    <p class="text-[11px] text-slate-500 mt-0.5">${kovanSayisi} Kayıtlı Kovan</p>
                                </div>
                            </div>
                            <div class="w-8 h-8 rounded-full bg-slate-100 border border-slate-200 flex items-center justify-center text-slate-400 group-hover:text-amber-600 transition">
                                <i data-lucide="chevron-right" class="w-4 h-4"></i>
                            </div>
                        </div>
                    `;
                });
                kategoriContainer.innerHTML = html;
                if (window.lucide) window.lucide.createIcons();
            },

            kategoriSec(id) {
                this.aktifKategoriId = id;
                const kat = this.data.kategoriler.find(k => k.id === id);
                if (!kat) return;

                document.getElementById('aktifKategoriBaslik').textContent = kat.isim;
                document.getElementById('kategoriAnaGorunum').classList.add('hidden');
                document.getElementById('kategoriDetayGorunum').classList.remove('hidden');

                this.aktifKovanlariCiz(kat.kovanlar);
            },

            kategoriGorunumuneDon() {
                this.aktifKategoriId = null;
                document.getElementById('kategoriDetayGorunum').classList.add('hidden');
                document.getElementById('kategoriAnaGorunum').classList.remove('hidden');
                this.arayuzuGuncelle();
            },

            aktifKovanlariCiz(kovanlar) {
                const container = document.getElementById('aktifKovanlarContainer');
                if (!kovanlar || kovanlar.length === 0) {
                    container.innerHTML = `<p class="text-center py-12 text-slate-400 text-xs">Bu kategoride henüz kovan bulunmuyor.</p>`;
                    return;
                }

                let html = '';
                kovanlar.forEach(k => {
                    const sonKontrol = k.gecmisKontroller && k.gecmisKontroller.length > 0 ? k.gecmisKontroller[k.gecmisKontroller.length - 1] : null;
                    const sonBal = sonKontrol ? `${sonKontrol.bal} kg` : '0 kg';
                    const sonNot = sonKontrol ? (sonKontrol.not.length > 45 ? sonKontrol.not.substring(0, 42) + '...' : sonKontrol.not) : 'Kayıt yok.';

                    html += `
                        <div class="p-4 rounded-2xl border border-slate-200 bg-white flex items-center justify-between hover:border-amber-400 transition cursor-pointer shadow-2xs" onclick="uygulama.kovanDetayModaliniAc(${k.id})">
                            <div class="flex-1 pr-3">
                                <div class="flex items-center gap-2">
                                    <h4 class="font-bold text-sm text-slate-800">${k.isim}</h4>
                                    <span class="text-[10px] font-bold bg-amber-100 text-amber-800 px-2.5 py-0.5 rounded-lg">Bal: ${sonBal}</span>
                                    <span class="text-[10px] font-semibold bg-slate-100 text-slate-600 px-2 py-0.5 rounded-lg">Ana: ${k.anaAri || '—'}</span>
                                </div>
                                <p class="text-xs text-slate-600 mt-1 line-clamp-1">${sonNot}</p>
                            </div>
                            <div class="flex items-center gap-1">
                                <button type="button" onclick="event.stopPropagation(); uygulama.kovanSil(${k.id})" class="text-slate-400 hover:text-rose-500 p-2 transition rounded-lg hover:bg-rose-50" title="Sil">
                                    <i data-lucide="trash-2" class="w-4 h-4"></i>
                                </button>
                            </div>
                        </div>
                    `;
                });
                container.innerHTML = html;
                if (window.lucide) window.lucide.createIcons();
            },

            kovanDetayModaliniAc(kovanId) {
                const kat = this.data.kategoriler.find(k => k.id === this.aktifKategoriId);
                if (!kat) return;
                const kovan = kat.kovanlar.find(k => k.id === kovanId);
                if (!kovan) return;

                this.aktifDuzenlenenKovanId = kovanId;

                document.getElementById('editKovanId').value = kovan.id;
                document.getElementById('detayIsim').value = kovan.isim || "";
                document.getElementById('detayAnaAri').value = kovan.anaAri || "";
                document.getElementById('yeniBal').value = "";
                document.getElementById('yeniHastalik').value = "";
                document.getElementById('yeniNot').value = "";

                this.gecmisKontrolleriCiz(kovan.gecmisKontroller);
                this.gelisimGrafigiCiz(kovan.gecmisKontroller);

                document.getElementById('kovanDetayModal').classList.remove('hidden');
                if (window.lucide) window.lucide.createIcons();
            },

            kovanDetayModaliniKapat() {
                const kat = this.data.kategoriler.find(k => k.id === this.aktifKategoriId);
                if(kat && this.aktifDuzenlenenKovanId) {
                     const kovan = kat.kovanlar.find(k => k.id === this.aktifDuzenlenenKovanId);
                     if(kovan) {
                         kovan.isim = document.getElementById('detayIsim').value.trim() || kovan.isim;
                         kovan.anaAri = document.getElementById('detayAnaAri').value.trim() || kovan.anaAri;
                         this.veriKaydet();
                     }
                }

                document.getElementById('kovanDetayModal').classList.add('hidden');
                if (kat) this.aktifKovanlariCiz(kat.kovanlar);
                this.arayuzuGuncelle();
            },

            yeniKontrolEkle() {
                const isim = document.getElementById('detayIsim').value.trim();
                const anaAri = document.getElementById('detayAnaAri').value.trim();
                const balInput = document.getElementById('yeniBal').value;
                const bal = balInput !== "" ? parseFloat(balInput) : null;
                const hastalik = document.getElementById('yeniHastalik').value.trim() || "Yok";
                const not = document.getElementById('yeniNot').value.trim() || "Kontrol gerçekleştirildi.";
                const tarih = new Date().toISOString().split('T')[0];

                const kat = this.data.kategoriler.find(k => k.id === this.aktifKategoriId);
                if (!kat) return;
                const kovan = kat.kovanlar.find(k => k.id === this.aktifDuzenlenenKovanId);
                if (!kovan) return;

                if (isim) kovan.isim = isim;
                if (anaAri) kovan.anaAri = anaAri;

                if (bal !== null) {
                    if (!kovan.gecmisKontroller) kovan.gecmisKontroller = [];
                    kovan.gecmisKontroller.push({ tarih, bal, hastalik, not });
                }

                document.getElementById('yeniBal').value = "";
                document.getElementById('yeniHastalik').value = "";
                document.getElementById('yeniNot').value = "";

                this.veriKaydet(); 

                this.gecmisKontrolleriCiz(kovan.gecmisKontroller);
                this.gelisimGrafigiCiz(kovan.gecmisKontroller);
            },

            gecmisKontrolleriCiz(kontroller) {
                const container = document.getElementById('gecmisKontrollerListesi');
                if (!kontroller || kontroller.length === 0) {
                    container.innerHTML = `<p class="text-center py-4 text-slate-400 text-xs">Henüz kontrol kaydı yok.</p>`;
                    return;
                }

                let html = '';
                [...kontroller].reverse().forEach((k, idx) => {
                    const gercekIdx = kontroller.length - 1 - idx;
                    let trendIkon = '<i data-lucide="minus" class="w-3.5 h-3.5 text-slate-400"></i>';
                    if(gercekIdx > 0) {
                        const onceki = kontroller[gercekIdx - 1].bal;
                        if(k.bal > onceki) trendIkon = '<i data-lucide="trending-up" class="w-3.5 h-3.5 text-emerald-500"></i>';
                        else if(k.bal < onceki) trendIkon = '<i data-lucide="trending-down" class="w-3.5 h-3.5 text-rose-500"></i>';
                    }

                    html += `
                        <div class="p-3 rounded-xl border border-slate-200 bg-white flex items-center justify-between text-xs shadow-2xs">
                            <div>
                                <div class="flex items-center gap-2 font-bold text-slate-800">
                                    <span>${k.tarih}</span>
                                    <span class="text-amber-600 bg-amber-50 px-2 py-0.5 rounded">${k.bal} kg</span>
                                    <span class="text-slate-500 font-normal">(${k.hastalik})</span>
                                </div>
                                <p class="text-slate-600 mt-1">${k.not}</p>
                            </div>
                            <div class="flex items-center gap-2 shrink-0">
                                ${trendIkon}
                                <button type="button" onclick="uygulama.kontrolSil(${gercekIdx})" class="text-slate-300 hover:text-rose-500 p-1"><i data-lucide="x" class="w-3.5 h-3.5"></i></button>
                            </div>
                        </div>
                    `;
                });
                container.innerHTML = html;
                if (window.lucide) window.lucide.createIcons();
            },

            kontrolSil(idx) {
                const kat = this.data.kategoriler.find(k => k.id === this.aktifKategoriId);
                if (!kat) return;
                const kovan = kat.kovanlar.find(k => k.id === this.aktifDuzenlenenKovanId);
                if (!kovan) return;

                kovan.gecmisKontroller.splice(idx, 1);
                
                this.veriKaydet(); 
                
                this.gecmisKontrolleriCiz(kovan.gecmisKontroller);
                this.gelisimGrafigiCiz(kovan.gecmisKontroller);
            },

            gelisimGrafigiCiz(kontroller) {
                const container = document.getElementById('gelisimGrafigi');
                if (!kontroller || kontroller.length === 0) {
                    container.innerHTML = `<p class="text-center text-xs text-slate-400 w-full m-auto">Grafik için kayıtlı bal verisi yok.</p>`;
                    return;
                }

                const maxBal = Math.max(...kontroller.map(k => Number(k.bal) || 0), 10);
                let html = '';

                kontroller.forEach((k, idx) => {
                    const yuzde = Math.max(15, Math.round((Number(k.bal) / maxBal) * 100));
                    let renk = "bg-amber-400";
                    if (idx > 0) {
                        const onceki = Number(kontroller[idx - 1].bal);
                        if (k.bal > onceki) renk = "bg-emerald-500";
                        else if (k.bal < onceki) renk = "bg-rose-500";
                    }

                    html += `
                        <div class="flex flex-col items-center gap-1 flex-1 h-full justify-end group relative">
                            <span class="text-[10px] font-bold text-slate-700">${k.bal}kg</span>
                            <div style="height: ${yuzde}%" class="w-full ${renk} rounded-t-lg transition-all duration-500"></div>
                            <span class="text-[9px] text-slate-500 truncate w-full text-center">${k.tarih.slice(5)}</span>
                        </div>
                    `;
                });
                container.innerHTML = html;
            },

            kontrolGecmisiniSifirla() {
                if(!confirm("Tüm geçmiş kontrol kayıtları sıfırlansın mı?")) return;
                const kat = this.data.kategoriler.find(k => k.id === this.aktifKategoriId);
                if (!kat) return;
                const kovan = kat.kovanlar.find(k => k.id === this.aktifDuzenlenenKovanId);
                if (!kovan) return;

                kovan.gecmisKontroller = [];
                
                this.veriKaydet();
                
                this.gecmisKontrolleriCiz(kovan.gecmisKontroller);
                this.gelisimGrafigiCiz(kovan.gecmisKontroller);
            },

            kovanSil(kovanId) {
                if(!confirm("Bu kovanı tamamen silmek istediğinize emin misiniz?")) return;
                const kat = this.data.kategoriler.find(k => k.id === this.aktifKategoriId);
                if (kat) {
                    kat.kovanlar = kat.kovanlar.filter(k => k.id !== kovanId);
                    
                    this.veriKaydet(); 
                    
                    this.aktifKovanlariCiz(kat.kovanlar);
                    this.arayuzuGuncelle();
                }
            },

            konumModaliniAc() {
                document.getElementById('sehirInput').value = "";
                document.getElementById('konumModal').classList.remove('hidden');
                document.getElementById('sehirInput').focus();
            },

            konumModaliniKapat() {
                document.getElementById('konumModal').classList.add('hidden');
            },

            async sehirAraVeGuncelle() {
                const aranan = document.getElementById('sehirInput').value.trim();
                if (!aranan) return;
                this.konumModaliniKapat();
                try {
                    const res = await fetch(`https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(aranan)}&count=1&language=tr&format=json`);
                    const data = await res.json();
                    if (data && data.results && data.results.length > 0) {
                        const yer = data.results[0];
                        await this.havaDurumunuYukle(yer.latitude, yer.longitude, yer.name);
                    } else {
                        alert("Aranan konum bulunamadı.");
                    }
                } catch(e) {
                    alert("Arama hatası.");
                }
            },

            yeniKategoriModaliniAc() {
                document.getElementById('kategoriIsimInput').value = "";
                document.getElementById('kategoriModal').classList.remove('hidden');
                document.getElementById('kategoriIsimInput').focus();
            },

            yeniKategoriModaliniKapat() {
                document.getElementById('kategoriModal').classList.add('hidden');
            },

            kategoriKaydet() {
                const isim = document.getElementById('kategoriIsimInput').value.trim();
                if (!isim) return;
                this.data.kategoriler.push({ id: Date.now(), isim: isim, kovanlar: [] });
                
                this.veriKaydet(); 
                
                this.yeniKategoriModaliniKapat();
                this.arayuzuGuncelle();
            },

            aktifKategoriyeKovanEkleModaliniAc() {
                const yeniId = Date.now();
                const kat = this.data.kategoriler.find(k => k.id === this.aktifKategoriId);
                if (!kat) return;
                const yeniKovan = {
                    id: yeniId,
                    isim: "Kovan #" + (kat.kovanlar.length + 1),
                    anaAri: new Date().getFullYear().toString(),
                    gecmisKontroller: [
                        { tarih: new Date().toISOString().split('T')[0], bal: 0, hastalik: "Yok", not: "İlk kurulum kaydı." }
                    ]
                };
                kat.kovanlar.push(yeniKovan);
                
                this.veriKaydet(); 
                
                this.aktifKovanlariCiz(kat.kovanlar);

            },

            sayfaDegistir(sayfaId) {
                if(sayfaId === 'Kovanlar') this.kategoriGorunumuneDon();
                ['AnaSayfa', 'Kovanlar', 'Saglik', 'Akademi'].forEach(s => {
                    const el = document.getElementById('sayfa' + s);
                    const nav = document.getElementById('nav' + s);
                    if (s.toLowerCase() === sayfaId.toLowerCase()) {
                        el?.classList.remove('hidden');
                        nav?.classList.add('nav-btn-active');
                        if (s === 'Akademi') this.rssYukle();
                    } else {
                        el?.classList.add('hidden');
                        nav?.classList.remove('nav-btn-active');
                    }
                });
                window.scrollTo({ top: 0, behavior: 'smooth' });
            },

            async rssYukle() {
                const container = document.getElementById('makaleListesiContainer');
                container.innerHTML = `<p class="text-center py-12 text-slate-400 text-xs">Yayınlar yükleniyor...</p>`;
                try {
                    const targetUrl = encodeURIComponent("https://www.polinasyon.org/atom.xml?redirect=false&start-index=1&max-results=50");
                    const res = await fetch(`https://api.rss2json.com/v1/api.json?rss_url=${targetUrl}`);
                    const data = await res.json();
                    if (data && data.items && data.items.length > 0) {
                        let html = '';
                        data.items.forEach(item => {
                            const descClean = item.description ? item.description.replace(/<[^>]*>?/gm, '').substring(0, 200) : "Detay bulunmuyor.";
                            html += `
                                <div class="rounded-2xl border border-slate-200 bg-white overflow-hidden mb-3">
                                    <div onclick="this.nextElementSibling.classList.toggle('active')" class="p-4 cursor-pointer flex justify-between items-center hover:bg-slate-50 transition">
                                        <h4 class="font-semibold text-sm text-slate-800">${item.title}</h4>
                                        <span class="text-xs text-slate-400">▼</span>
                                    </div>
                                    <div class="accordion-panel px-4 pb-4 pt-2 border-t border-slate-100 text-xs text-slate-600">
                                        <p class="mb-3">${descClean}...</p>
                                        <a href="${item.link}" target="_blank" class="text-amber-600 font-bold underline inline-block">Devamını Oku →</a>
                                    </div>
                                </div>
                            `;
                        });
                        container.innerHTML = html;
                    }
                } catch(e) {
                    container.innerHTML = `<p class="text-center py-6 text-slate-400 text-xs">Yayınlar yüklenemedi. <a href="https://www.polinasyon.org" target="_blank" class="text-amber-600 underline">Ziyaret et</a></p>`;
                }
            },

            verileriYedekle() {
                const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(this.data));
                const dl = document.createElement('a');
                dl.setAttribute("href", dataStr);
                dl.setAttribute("download", "polinasyon_yedek.json");
                dl.click();
            }
        };

        window.onload = () => {
            uygulama.veriYukle(); 
            uygulama.arayuzuGuncelle();
            uygulama.havaDurumunuYukle(uygulama.data.lat, uygulama.data.lon, uygulama.data.konum);
            if (window.lucide) window.lucide.createIcons();
        };
